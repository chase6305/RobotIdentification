cmake_minimum_required(VERSION 3.15)
project(robot_identification_synthetic_benchmarks LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  message(FATAL_ERROR "Allocator instrumentation requires Linux and a GNU-compatible linker")
endif()
find_package(robot_identification REQUIRED)
option(ROBOT_IDENTIFICATION_BENCH_PINOCCHIO "Build the optional Pinocchio dataset benchmark" OFF)
# The linked core exports its C++17 requirement to all consumers.
foreach(name IN ITEMS core_bench csv_bench)
  add_executable(${name} ${name}.cpp)
  target_link_libraries(${name} PRIVATE robot_identification::robot_identification)
  target_link_options(${name} PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc)
endforeach()
add_executable(preprocessing_bench preprocessing_bench.cpp)
target_link_libraries(preprocessing_bench PRIVATE robot_identification::robot_identification)
if(robot_identification_VERSION VERSION_GREATER_EQUAL "0.2.0")
  add_executable(design_bench design_bench.cpp)
  target_link_libraries(design_bench PRIVATE robot_identification::robot_identification)
  add_executable(parameter_bench parameter_bench.cpp)
  target_link_libraries(parameter_bench PRIVATE robot_identification::robot_identification)
  target_link_options(parameter_bench PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc)
  if(robot_identification_VERSION VERSION_GREATER_EQUAL "0.2.4")
    target_compile_definitions(design_bench PRIVATE ROBOT_IDENTIFICATION_BENCH_HAS_EVALUATE_INTO)
  endif()
endif()

option(ROBOT_IDENTIFICATION_BENCH_EXACT_DOT_REFERENCE "Build the independent Boost integer coordinate oracle" OFF)
if(ROBOT_IDENTIFICATION_BENCH_EXACT_DOT_REFERENCE)
  if(robot_identification_VERSION VERSION_LESS "0.2.0")
    message(FATAL_ERROR "The coordinate oracle requires core >=0.2.0")
  endif()
  find_package(Boost REQUIRED)
  add_executable(parameter_dot_stress parameter_dot_stress.cpp)
  target_include_directories(parameter_dot_stress SYSTEM PRIVATE ${Boost_INCLUDE_DIRS})
  target_link_libraries(parameter_dot_stress PRIVATE robot_identification::robot_identification)
endif()
if(ROBOT_IDENTIFICATION_BENCH_PINOCCHIO)
  find_package(robot_identification_pinocchio REQUIRED)
  add_executable(pinocchio_bench pinocchio_bench.cpp)
  target_link_libraries(pinocchio_bench PRIVATE robot_identification::pinocchio)
  target_link_options(pinocchio_bench PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc)
  if(robot_identification_pinocchio_VERSION VERSION_GREATER_EQUAL "0.2.0")
    target_compile_definitions(pinocchio_bench PRIVATE ROBOT_IDENTIFICATION_BENCH_HAS_BATCH_API)
  endif()
endif()

option(ROBOT_IDENTIFICATION_BENCH_STRESS_TARGETS "Build opt-in numerical and storage campaigns" OFF)
if(ROBOT_IDENTIFICATION_BENCH_STRESS_TARGETS)
  foreach(name IN ITEMS hybrid_rollout_demo solver_stress solver_scale_stress
      csv_stress preprocessing_stress metrics_stress robust_stress
      trajectory_stress design_stress)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE robot_identification::robot_identification)
  endforeach()
  if(ROBOT_IDENTIFICATION_BENCH_PINOCCHIO)
    find_package(Threads REQUIRED)
    add_executable(pinocchio_stress pinocchio_stress.cpp)
    target_link_libraries(pinocchio_stress PRIVATE robot_identification::pinocchio Threads::Threads)
  endif()
endif()
